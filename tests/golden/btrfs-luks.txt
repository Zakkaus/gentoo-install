[partition]
  wipe existing signatures from disk
  create a gpt partition table on disk as table
  create partition 1 on disk as esp: esp, 512MiB
  create partition 2 on disk as cryptroot: data, the rest of the disk
  reread the partition table of disk and wait for its nodes

[array]
  format cryptroot as LUKS2 and open it as /dev/mapper/root

[format]
  make a vfat filesystem on esp as espfs labelled ESP
  make a btrfs filesystem on root-luks as rootfs
  create btrfs subvolume @home on root-luks as sub-home
  create btrfs subvolume @ on root-luks as sub-root

[mount]
  mount root-luks at / with compress=zstd:1,subvol=@
  mount esp at /efi with umask=0077
  mount root-luks at /home with compress=zstd:1,subvol=@home

[stage3]
  download the newest systemd stage3 from https://distfiles.gentoo.org, verify it against BB572E0E2D182910 and unpack it into the target

[chroot]
  mount proc, sys, dev and run into the target and copy resolv.conf

[portage]
  write /etc/portage/make.conf with COMMON_FLAGS, CFLAGS, CXXFLAGS, FCFLAGS, FFLAGS, MAKEOPTS, USE, VIDEO_CARDS, ACCEPT_LICENSE, L10N, FEATURES; 3 mirrors fastest first, measured
  create the three autounmask files emerge writes its decisions into
  fetch the first ebuild repository snapshot with emerge-webrsync
  select profile default/linux/amd64/23.0/desktop/systemd
  install git, which every later repository sync needs: emerge dev-vcs/git
  point repository gentoo at https://mirrors.cernet.edu.cn/git/gentoo-portage.git, commit signatures verified
  sync repository gentoo
  point repository gentoo-zh at https://mirrors.cernet.edu.cn/gentoo-zh.git
  sync repository gentoo-zh
  accept ~amd64 for packages from gentoo-zh only
  run getuto so Portage has a keyring to verify binary packages against
  install the key the community binary packages are signed with: emerge sec-keys/openpgp-keys-gentoozh, from source
  import 38A0234EC16AD42E from /usr/share/openpgp-keys/gentoozh.asc and locally sign it
  add verified binary package host gentoo-zh at https://mirrors.cernet.edu.cn/gentoo-zh/binpkgs/x86-64

[system]
  generate and verify locales en_US.UTF-8, zh_TW.UTF-8
  set the system locale to zh_TW.UTF-8
  set the timezone to Asia/Taipei
  set the console keymap to us and its font to default8x16
  set the hostname to gig
  give the target its own machine id
  write /etc/fstab with entries for /, /home, /efi
  write /etc/crypttab for root
  create user zakk in users, wheel, audio, video, render, usb, input with no password
  lock the root account
  install sudo: emerge app-admin/sudo
  let the wheel group run sudo, with a password
  install sshd: emerge net-misc/openssh
  enable sshd.service at boot
  configure the wired interface for DHCP
  enable systemd-networkd.service at boot

[kernel]
  set sys-kernel/installkernel to use dracut
  install the hook that puts a kernel in /boot: emerge sys-kernel/installkernel
  tell dracut to carry crypt, btrfs
  accept the linux-firmware licence
  install the initramfs builder and firmware: emerge sys-kernel/dracut sys-kernel/linux-firmware
  install the storage tools: emerge sys-fs/cryptsetup sys-fs/btrfs-progs sys-fs/dosfstools
  install the kernel: emerge sys-kernel/gentoo-cjk-sources, from source
  point /usr/src/linux at the kernel that was just unpacked
  configure the kernel from defconfig with no extra options
  build the kernel and its modules, then install both

[bootloader]
  install the bootloader: emerge sys-boot/grub sys-boot/efibootmgr
  write /etc/default/grub with cmdline quiet
  install GRUB for uefi on the esp at /efi

[packages]
  install the plasma group: emerge kde-plasma/plasma-meta x11-base/xorg-server kde-apps/konsole kde-apps/dolphin
  enable sddm.service at boot
  install the fcitx5-rime group: emerge app-i18n/fcitx app-i18n/fcitx-rime app-i18n/fcitx-configtool app-i18n/rime-data
  install the noto-cjk group: emerge media-fonts/noto-cjk media-fonts/noto media-libs/fontconfig
  install the flclash group: emerge net-proxy/flclash

[finish]
  unmount everything under the target
