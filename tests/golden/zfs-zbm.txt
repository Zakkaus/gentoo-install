[partition]
  wipe existing signatures from disk
  create a gpt partition table on disk as table
  create partition 1 on disk as esp: esp, 512MiB
  create partition 2 on disk as poolpart: data, the rest of the disk
  reread the partition table of disk and wait for its nodes

[format]
  make a vfat filesystem on esp as espfs labelled ESP

[zfs]
  create zpool zpcala as pool from poolpart with native encryption
  create dataset zpcala/ROOT/gentoo/home as ds-home, mountpoint /home
  create dataset zpcala/ROOT/gentoo/root as ds-root, mountpoint /

[mount]
  mount dataset zpcala/ROOT/gentoo/root at /
  mount esp at /efi with umask=0077
  mount dataset zpcala/ROOT/gentoo/home at /home

[stage3]
  download the newest systemd stage3 from https://distfiles.gentoo.org, verify it against BB572E0E2D182910 and unpack it into the target

[chroot]
  mount proc, sys, dev and run into the target and copy resolv.conf
  mount efivarfs in the target so efibootmgr can write a boot entry

[portage]
  write /etc/portage/make.conf with COMMON_FLAGS, CFLAGS, CXXFLAGS, FCFLAGS, FFLAGS, MAKEOPTS, ACCEPT_LICENSE, L10N, FEATURES; 2 mirrors in the configured order
  create the three autounmask files emerge writes its decisions into
  fetch the first ebuild repository snapshot with emerge-webrsync
  select profile default/linux/amd64/23.0/systemd
  install git, which every later repository sync needs: emerge dev-vcs/git
  point repository gentoo at https://github.com/gentoo-mirror/gentoo.git, commit signatures verified
  sync repository gentoo

[system]
  generate and verify locales en_US.UTF-8, zh_TW.UTF-8
  set the system locale to zh_TW.UTF-8
  set the timezone to Asia/Taipei
  set the console keymap to us and its font to latarcyrheb-sun32
  set the hostname to zfsbox
  give the target its own machine id
  write /etc/fstab with entries for /efi
  create user zakk in users, wheel, audio, video, render, usb, input with a password
  lock the root account
  let the wheel group run sudo, with a password
  install the network tools: emerge net-misc/dhcpcd
  configure the wired interface for DHCP
  enable systemd-networkd.service at boot

[kernel]
  set sys-kernel/installkernel to use dracut
  tell dracut to carry zfs
  accept the linux-firmware licence
  install the initramfs builder and firmware: emerge sys-kernel/dracut sys-kernel/linux-firmware
  install the storage tools: emerge sys-fs/zfs sys-fs/dosfstools
  install the kernel: emerge sys-kernel/gentoo-cjk-sources, from source

[bootloader]
  install the bootloader: emerge sys-boot/zfsbootmenu sys-boot/efibootmgr
  write /etc/hostid so the pool imports on the target
  build ZFSBootMenu into /efi/EFI/zbm/vmlinuz.EFI and boot zpcala/ROOT/gentoo/root from it

[packages]
  install the console group: emerge sys-apps/kbd app-editors/vim

[finish]
  unmount everything under the target and export zpcala
